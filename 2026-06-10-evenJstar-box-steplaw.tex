\noindent\textbf{Setup.} $\lambda\vdash n=2m$; $G_\lambda(i)=\langle s_\lambda,\psi^m\rangle$,
$\psi=h_2+ie_2$, is the fiber at $\zeta_4=i$ of $G_\lambda(q)=\sum_{T\in\mathrm{SYT}(\lambda)}q^{s(T)}$.
Write $\pi=1+i$, so $\Z[i]/(\pi)\cong\F_2$, $v_\pi(N)=2v_2(N)$ for $N\in\Z$, $\pi^2=2i$. With
$h_2=\tfrac12(p_1^2+p_2)$, $e_2=\tfrac12(p_1^2-p_2)$ one has $p_1^2=p_2+2e_2$; set
$\chi_k:=\chi^\lambda(2^k1^{2m-2k})=\langle s_\lambda,p_2^kp_1^{2m-2k}\rangle$, $f^\lambda=\chi_0$.

\section{Reformulation}
\begin{lemma} $\psi=p_2+\pi e_2$, hence
$G_\lambda(i)=\Psi(\pi)$ where $\Psi(Y)=\sum_{r=0}^m\binom mr R_rY^r$,
$R_r=\langle s_\lambda,p_2^{m-r}e_2^r\rangle\in\Z$, $R_0=\chi^\lambda(2^m)$.
\end{lemma}
\begin{proof} $h_2=p_2+e_2$ gives $\psi=p_2+(1+i)e_2$; expand $\psi^m$ and pair with $s_\lambda$.\end{proof}
The term valuations are $\val(r)=r+2v_2(\binom mr R_r)\ge0$, a $\pi$-adic Newton polygon for $G$
(minimum locus $=J^*$). Also $p_1^2-ip_2=-i\pi(p_2+\pi e_2)$, so with
$\Phi(z):=\langle s_\lambda,(p_1^2+zp_2)^m\rangle=\sum_k\binom mk\chi_k z^k$,
$G_\lambda(i)=(\tfrac{1+i}2)^m\Phi(-i)$ and $G_\lambda(i)=0\iff(z^2+1)\mid\Phi(z)$ in $\Z[z]$.

\section{A one-line congruence}
\begin{proposition} $G_\lambda(i)\equiv\chi^\lambda(2^m)\pmod\pi$. \end{proposition}
\begin{proof} In $\Psi(\pi)=\sum_r\binom mrR_r\pi^r$ every $r\ge1$ term is $\equiv0\pmod\pi$, leaving
$R_0=\chi^\lambda(2^m)$. \end{proof}
\begin{lemma} $\chi_k\equiv f^\lambda\pmod2$ for all $k$. \end{lemma}
\begin{proof} $p_1^2\equiv p_2\pmod2$, so $p_2^kp_1^{2m-2k}\equiv p_1^{2m}\pmod2$; pair with the
integral $s_\lambda$. \end{proof}
\begin{corollary} $\chi^\lambda(2^m)\equiv f^\lambda\pmod2$; thus $f^\lambda$ odd $\Rightarrow
G_\lambda(i)\not\equiv0\pmod\pi\Rightarrow G_\lambda(i)\neq0$, and every tie $(|J^*|>1)$ has
$f^\lambda$ even. \end{corollary}

\section{The 2-adic engine}
\begin{theorem} $\displaystyle \Phi(z)=\sum_{r=0}^m\binom mr 2^r R_r\,(1+z)^{m-r}.$ \end{theorem}
\begin{proof} $p_1^2+zp_2=(1+z)p_2+2e_2$; expand and pair with $s_\lambda$. \end{proof}
\begin{corollary} $\Phi(z)\equiv\chi^\lambda(2^m)(1+z)^m\pmod2$. If $\chi^\lambda(2^m)$ is odd, the
coarse ($z=-i$, a unit) Newton locus $B^*=\arg\min_kv_2(\binom mk\chi_k)$ equals the set of binary
submasks of $m$ (Lucas), so $|B^*|=2^{s_2(m)}$. \end{corollary}

\section{The box, and the reduction}
For a polynomial evaluated at a unit, the minimal-valuation locus is an affine 2-adic box
$j_0+\{\sum_{a\in S}2^a\}$ iff, after dividing by $2^{e}$ ($e=\min v_2$ of coefficients), the mod-2
reduction is $z^{j_0}(1+z)^{g}$ (over $\F_2$, $(1+z)^g=\prod_{a\in S}(1+z^{2^a})$ has exactly that
support). The sharp polygon $J^*$ lies in one parity class (Fact 4.1); restricting to it and
absorbing the parity tilt $2^s$ gives a scaled polynomial whose bottom-mod-2 part is the analog.
\begin{conjecture}[Box] Coarse: $\Phi(z)/2^{e}\equiv z^{j_0}(1+z)^{g}\pmod2$. Sharp: the
parity-restricted tilt-scaled polynomial is $\equiv y^{l_0}(1+y)^{g'}\pmod2$. Hence $B^*,J^*$ are
affine 2-adic boxes, $|J^*|$ is a power of two, and even on a tie.
\end{conjecture}
The engine (Cor.~7) proves the $r=0$ layer of Theorem~6. When $\chi^\lambda(2^m)$ is even (exactly
the tie case) that layer vanishes mod 2 and the next layer involves $e_2\bmod2$, where
$p_1^2\equiv p_2$ no longer collapses the two generators; a uniform proof needs the statement
mod $\pi^k$. Verified exhaustively for all $\lambda\vdash2m$, $m\le12$: coarse $4114/4114$, sharp
$1624/1624$ ties; $|B^*|,|J^*|\in\{1,2,4,8\}$.

\medskip\noindent\textbf{Honest scope.} This does not close non-vanishing: on a tie the leading
$\pi$-coefficient cancels and $v_\pi(G)$ is set by an interleaved cancellation tower of unbounded
depth ($d=v_\pi(G)-\mu$ ranges $1$--$22$, $m\le12$). The unique total vanisher is $(2,2)$, where
$\Phi=2(1+z^2)$.
